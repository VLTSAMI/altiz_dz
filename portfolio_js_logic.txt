        const grid = document.getElementById('portfolio-grid');
        const filterBtns = document.querySelectorAll('.filter-btn');
        const DRIVE_URL = 'https://script.google.com/macros/s/AKfycbyNWWFcRwGwEB5F4E36QRwL5mVxwXMlp9IBXTzub4c9bWaPs9UUTzM6vmCVDxeQnNwv1w/exec';

        function toggleDebug() {
            const d = document.getElementById('debug-info');
            d.style.display = d.style.display === 'none' ? 'block' : 'none';
        }

        function fetchJSONP(url) {
            return new Promise((resolve, reject) => {
                const callbackName = 'jsonp_' + Date.now();
                window[callbackName] = (data) => {
                    delete window[callbackName];
                    const s = document.getElementById(callbackName);
                    if (s) s.remove();
                    resolve(data);
                };
                const script = document.createElement('script');
                script.id = callbackName;
                script.src = url + (url.indexOf('?') >= 0 ? '&' : '?') + 'callback=' + callbackName;
                script.onerror = () => reject(new Error("Script Load Failed"));
                document.body.appendChild(script);
                setTimeout(() => { 
                    if (window[callbackName]) {
                        delete window[callbackName];
                        reject(new Error("Timeout"));
                    }
                }, 15000);
            });
        }

        async function loadProjects(categoryFilter = 'all') {
            grid.innerHTML = '<div style="grid-column: 1/-1; text-align: center; padding: 4rem; color: var(--primary-accent);">Loading Portfolio...</div>';
            try {
                const allProjects = await fetchJSONP(DRIVE_URL);
                console.log("Data:", allProjects);
                document.getElementById('debug-info').innerText = JSON.stringify(allProjects, null, 2);
                
                grid.innerHTML = '';
                let count = 0;

                allProjects.forEach(p => {
                    if (categoryFilter !== 'all' && p.category.toLowerCase() !== categoryFilter.toLowerCase()) return;
                    count++;
                    
                    let mediaHtml = '';
                    if (p.type === 'video') {
                        let src = p.path.replace('view?export=download', 'preview').replace('view?usp=sharing', 'preview');
                        if (!src.includes('preview')) src = src.replace('uc?export=view&id=', 'file/d/').split('&')[0] + '/preview';
                        mediaHtml = `<iframe src="${src}" style="width: 100%; height: 100%; border: none;" allow="autoplay"></iframe>`;
                    } else {
                        mediaHtml = `<img src="${p.path}" style="width: 100%; height: 100%; object-fit: cover;">`;
                    }

                    const card = document.createElement('div');
                    card.className = 'portfolio-card active';
                    card.innerHTML = `
                        ${mediaHtml}
                        <div class="portfolio-overlay" style="opacity: 1; background: linear-gradient(to top, rgba(0,0,0,0.9), transparent);">
                            <span class="portfolio-category">${p.category}</span>
                            <h3 class="portfolio-title">${p.title}</h3>
                            <a href="${p.path}" target="_blank" class="portfolio-link">View Project</a>
                        </div>
                    `;
                    grid.appendChild(card);
                });

                if (count === 0) grid.innerHTML = '<div style="grid-column: 1/-1; text-align: center; padding: 4rem;">No projects found.</div>';
            } catch (e) {
                grid.innerHTML = `<div style="grid-column: 1/-1; text-align: center; padding: 4rem; color: red;">Sync Error: ${e.message}</div>`;
            }
        }

        const urlParams = new URLSearchParams(window.location.search);
        loadProjects(urlParams.get('filter') || 'all');

        filterBtns.forEach(btn => {
            btn.onclick = () => {
                filterBtns.forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                loadProjects(btn.dataset.filter);
            };
        });

        document.addEventListener('DOMContentLoaded', () => {
            const savedLang = localStorage.getItem('preferredLang') || 'en';
            if (savedLang === 'ar') document.body.classList.add('rtl-layout');
        });
